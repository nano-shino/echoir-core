# The built-in model catalogue: models/registry.json as a source file in the build tree. The JSON is
# cut into raw string literals of at most 4000 bytes, because MSVC limits one literal to 16 KB.
# The file is rewritten only when its content changes, and a change to the JSON reruns configure.
set(ECHOIR_REGISTRY_JSON "${PROJECT_SOURCE_DIR}/models/registry.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ECHOIR_REGISTRY_JSON}")
file(READ "${ECHOIR_REGISTRY_JSON}" ECHOIR_REGISTRY_TEXT)
string(FIND "${ECHOIR_REGISTRY_TEXT}" ")ECHOIR_JSON\"" ECHOIR_REGISTRY_DELIMITER_POS)
if(NOT ECHOIR_REGISTRY_DELIMITER_POS EQUAL -1)
  message(FATAL_ERROR "models/registry.json contains the raw string delimiter")
endif()
string(LENGTH "${ECHOIR_REGISTRY_TEXT}" ECHOIR_REGISTRY_LENGTH)
set(ECHOIR_REGISTRY_LITERALS "")
set(ECHOIR_REGISTRY_OFFSET 0)
while(ECHOIR_REGISTRY_OFFSET LESS ECHOIR_REGISTRY_LENGTH)
  string(SUBSTRING "${ECHOIR_REGISTRY_TEXT}" ${ECHOIR_REGISTRY_OFFSET} 4000 ECHOIR_REGISTRY_PIECE)
  string(APPEND ECHOIR_REGISTRY_LITERALS
    "    R\"ECHOIR_JSON(${ECHOIR_REGISTRY_PIECE})ECHOIR_JSON\"\n")
  math(EXPR ECHOIR_REGISTRY_OFFSET "${ECHOIR_REGISTRY_OFFSET} + 4000")
endwhile()
set(ECHOIR_BUILTIN_REGISTRY_TMP "${CMAKE_CURRENT_BINARY_DIR}/registry_builtin.cpp.tmp")
set(ECHOIR_BUILTIN_REGISTRY_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/registry_builtin.cpp")
file(WRITE "${ECHOIR_BUILTIN_REGISTRY_TMP}"
"// Generated by CMake from models/registry.json. Do not edit.
#include \"models/registry.h\"

namespace echoir::models {

std::string_view builtin_registry_json() {
    return
${ECHOIR_REGISTRY_LITERALS}    ;
}

}  // namespace echoir::models
")
configure_file("${ECHOIR_BUILTIN_REGISTRY_TMP}" "${ECHOIR_BUILTIN_REGISTRY_SOURCE}" COPYONLY)

# Pure-logic core: src/audio, src/text, src/stream, src/capture and src/models, except the device
# query, which needs ggml. No backend dependency, so the unit tests build with no models or
# submodules. Sources are globbed; CONFIGURE_DEPENDS picks up new files on the next build.
file(GLOB ECHOIR_PURE_SOURCES CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/audio/*.cpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/text/*.cpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/stream/*.cpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/capture/*.cpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/models/*.cpp")
list(FILTER ECHOIR_PURE_SOURCES EXCLUDE REGEX "/models/devices\\.cpp$")
list(APPEND ECHOIR_PURE_SOURCES "${ECHOIR_BUILTIN_REGISTRY_SOURCE}")

# miniaudio's Core Audio backend on iOS includes AVFoundation, which only compiles as Objective-C.
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set_source_files_properties(audio/miniaudio_impl.cpp PROPERTIES COMPILE_OPTIONS "-xobjective-c++")
endif()

add_library(echoir_pure STATIC ${ECHOIR_PURE_SOURCES})
target_include_directories(echoir_pure PUBLIC
  "${CMAKE_CURRENT_SOURCE_DIR}" "${PROJECT_SOURCE_DIR}/include")
target_link_libraries(echoir_pure PUBLIC echoir_single PRIVATE echoir_options)

# Model-backed core: src/vad (Silero through whisper_vad_*, FireRedVAD on ggml) and src/backends
# (IAsrBackend, WhisperBackend, QwenBackend). Built only when libwhisper is, because Silero needs
# it. qwen_*.cpp is left out when the Qwen backend is off; ECHOIR_HAS_QWEN tells the factory which
# one exists.
if(ECHOIR_BUILD_WHISPER)
  file(GLOB ECHOIR_ENGINE_SOURCES CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/vad/*.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/backends/*.cpp")
  list(APPEND ECHOIR_ENGINE_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/models/devices.cpp")
  if(ECHOIR_FAKE_BACKEND)
    # The scripted backend and VAD that the factories return in this build.
    file(GLOB ECHOIR_FAKE_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/fake/*.cpp")
    list(APPEND ECHOIR_ENGINE_SOURCES ${ECHOIR_FAKE_SOURCES})
  endif()
  if(NOT ECHOIR_BUILD_QWEN)
    list(FILTER ECHOIR_ENGINE_SOURCES EXCLUDE REGEX "/backends/qwen_[^/]*\\.cpp$")
  endif()
endif()
# Without any engine source there is no engine target, so a core-only build still configures.
if(ECHOIR_BUILD_WHISPER AND ECHOIR_ENGINE_SOURCES)
  add_library(echoir_engine STATIC ${ECHOIR_ENGINE_SOURCES})
  target_include_directories(echoir_engine PUBLIC
    "${CMAKE_CURRENT_SOURCE_DIR}" "${PROJECT_SOURCE_DIR}/include")
  # FireRedModel calls ggml directly (CPU backend, GGUF reader). whisper already links ggml; this
  # names the dependency.
  target_link_libraries(echoir_engine PUBLIC echoir_pure whisper PRIVATE echoir_options ggml)
  if(ECHOIR_BUILD_QWEN)
    target_link_libraries(echoir_engine PUBLIC llama mtmd)
    target_compile_definitions(echoir_engine PUBLIC ECHOIR_HAS_QWEN=1)
  endif()
  # Linked into the shared echoir library, so it needs PIC like echoir_pure.
  set_target_properties(echoir_engine PROPERTIES POSITION_INDEPENDENT_CODE ON)
endif()
